/** * TopBliss Events Calendar: Google Sheet -> public JSON feed. * * HOW TO USE * 1. Create a new Google Sheet (name it e.g. "TopBliss Calendar Data"). * 2. Extensions > Apps Script. Delete any code there and paste this whole file. Save. * 3. Run the function "setup" once (Run button). Approve the permission prompt. * It creates the three tabs (Events, Birthdays, Memos) with headers and dropdowns. * 4. Deploy > New deployment > type "Web app". * Execute as: Me * Who has access: Anyone * Click Deploy and copy the Web app URL (ends in /exec). * 5. Paste that URL into API_URL inside index.html. * * After you edit the code later, use Deploy > Manage deployments > Edit > New version, * so the same URL keeps working. * * Whatever is in these three tabs is public to anyone with the website link. * Don't put anything private in them. Share the Sheet itself only with HR. */ var SHEETS = { Events: [ "title", "category", "date", "end_date", "start_time", "end_time", "repeats", "repeat_until", "location", "image_url", "link", "details", "posted_by" ], Birthdays: ["name", "month", "day", "team"], Memos: ["title", "type", "date", "effective_date", "issued_by", "message", "link", "pinned"] }; var CATEGORY_LIST = ["Company Holiday", "Training / Academy", "Evaluation / Deadline", "Meeting", "Birthday / Anniversary"]; var REPEAT_LIST = ["Does not repeat", "Weekly", "Monthly", "Yearly"]; var MEMO_TYPE_LIST = ["Company Memo", "Code of Conduct", "Announcement", "Policy"]; var DATE_COLS = { date: 1, end_date: 1, repeat_until: 1, effective_date: 1 }; var TIME_COLS = { start_time: 1, end_time: 1 }; var CACHE_SECONDS = 30; /** Run once. Safe to run again: it only adds what's missing. */ function setup() { var ss = SpreadsheetApp.getActiveSpreadsheet(); Object.keys(SHEETS).forEach(function (name) { var sh = ss.getSheetByName(name) || ss.insertSheet(name); var headers = SHEETS[name]; if (sh.getLastRow() === 0) { sh.getRange(1, 1, 1, headers.length).setValues([headers]); } sh.getRange(1, 1, 1, headers.length) .setFontWeight("bold").setBackground("#E1EBDD").setFontColor("#1B2A1E"); sh.setFrozenRows(1); sh.setColumnWidths(1, headers.length, 150); }); var ev = ss.getSheetByName("Events"); setDropdown_(ev, "category", CATEGORY_LIST); setDropdown_(ev, "repeats", REPEAT_LIST); setFormat_(ev, "date", "yyyy-mm-dd"); setFormat_(ev, "end_date", "yyyy-mm-dd"); setFormat_(ev, "repeat_until", "yyyy-mm-dd"); setFormat_(ev, "start_time", "h:mm AM/PM"); setFormat_(ev, "end_time", "h:mm AM/PM"); setColWidth_(ev, "details", 320); setColWidth_(ev, "title", 240); var bd = ss.getSheetByName("Birthdays"); setColWidth_(bd, "name", 220); setNumberRule_(bd, "month", 1, 12); setNumberRule_(bd, "day", 1, 31); var mm = ss.getSheetByName("Memos"); setDropdown_(mm, "type", MEMO_TYPE_LIST); setFormat_(mm, "date", "yyyy-mm-dd"); setFormat_(mm, "effective_date", "yyyy-mm-dd"); setColWidth_(mm, "message", 480); setColWidth_(mm, "title", 260); var pinCol = colIndex_(mm, "pinned"); if (pinCol) mm.getRange(2, pinCol, 998, 1).insertCheckboxes(); var def = ss.getSheetByName("Sheet1"); if (def && def.getLastRow() === 0 && ss.getSheets().length > 1) ss.deleteSheet(def); SpreadsheetApp.getUi().alert( "Setup done. Add your rows to the Events, Birthdays and Memos tabs, then deploy as a Web app (see the notes at the top of this script)." ); } /** The public feed. */ function doGet() { try { var cache = CacheService.getScriptCache(); var hit = cache.get("feed"); if (hit) return json_(hit); var ss = SpreadsheetApp.getActiveSpreadsheet(); var tz = ss.getSpreadsheetTimeZone(); var out = { ok: true, updated: new Date().toISOString(), events: readSheet_(ss.getSheetByName("Events"), tz), birthdays: readSheet_(ss.getSheetByName("Birthdays"), tz), memos: readSheet_(ss.getSheetByName("Memos"), tz) }; var text = JSON.stringify(out); try { cache.put("feed", text, CACHE_SECONDS); } catch (e) { /* too big to cache: fine */ } return json_(text); } catch (err) { return json_(JSON.stringify({ ok: false, error: String(err) })); } } function json_(text) { return ContentService.createTextOutput(text).setMimeType(ContentService.MimeType.JSON); } function readSheet_(sh, tz) { if (!sh || sh.getLastRow() < 2) return []; var range = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()); var values = range.getValues(); var display = range.getDisplayValues(); var headers = values[0].map(function (h) { return String(h).trim().toLowerCase().replace(/[^a-z0-9]+/g, "_").replace(/^_|_$/g, ""); }); var rows = []; for (var r = 1; r < values.length; r++) { var obj = {}, any = false; for (var c = 0; c < headers.length; c++) { var key = headers[c]; if (!key) continue; var v = values[r][c]; if (TIME_COLS[key]) v = timeString_(v, display[r][c]); else if (DATE_COLS[key]) v = dateString_(v, tz); else if (v instanceof Date) v = Utilities.formatDate(v, tz, "yyyy-MM-dd"); if (v !== "" && v !== null && v !== false) any = true; obj[key] = v; } if (any) rows.push(obj); } return rows; } function dateString_(v, tz) { if (v instanceof Date) return Utilities.formatDate(v, tz, "yyyy-MM-dd"); var s = String(v).trim(); if (/^\d{4}-\d{1,2}-\d{1,2}$/.test(s)) { var p = s.split("-"); return p[0] + "-" + ("0" + p[1]).slice(-2) + "-" + ("0" + p[2]).slice(-2); } return ""; } /** Reads the cell as shown ("3:30 PM", "15:30") so odd 1899 date offsets can't skew it. */ function timeString_(v, shown) { var s = String(shown || v || "").trim(); var m = s.match(/^(\d{1,2}):(\d{2})(?::\d{2})?\s*([AaPp][Mm])?$/); if (!m) return ""; var h = parseInt(m[1], 10); if (m[3]) { var pm = m[3].toLowerCase() === "pm"; if (pm && h < 12) h += 12; if (!pm && h === 12) h = 0; } return ("0" + h).slice(-2) + ":" + m[2]; } /* ---------- setup helpers ---------- */ function colIndex_(sh, header) { var headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]; var i = headers.indexOf(header); return i < 0 ? 0 : i + 1; } function setDropdown_(sh, header, list) { var c = colIndex_(sh, header); if (!c) return; var rule = SpreadsheetApp.newDataValidation().requireValueInList(list, true).setAllowInvalid(false).build(); sh.getRange(2, c, 998, 1).setDataValidation(rule); } function setFormat_(sh, header, fmt) { var c = colIndex_(sh, header); if (c) sh.getRange(2, c, 998, 1).setNumberFormat(fmt); } function setColWidth_(sh, header, w) { var c = colIndex_(sh, header); if (c) sh.setColumnWidth(c, w); } function setNumberRule_(sh, header, min, max) { var c = colIndex_(sh, header); if (!c) return; var rule = SpreadsheetApp.newDataValidation().requireNumberBetween(min, max) .setHelpText("Enter a number from " + min + " to " + max).setAllowInvalid(false).build(); sh.getRange(2, c, 998, 1).setDataValidation(rule); }
